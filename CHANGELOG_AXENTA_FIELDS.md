## 📊 Примеры использования

### SQL запросы

```sql
-- Количество объектов с информацией о создателе
SELECT COUNT(*) FROM axenta_object_snapshots 
WHERE creator_name IS NOT NULL;

-- Объекты, созданные за последний месяц в Axenta
SELECT * FROM axenta_object_snapshots 
WHERE axenta_created_at >= NOW() - INTERVAL '1 month';

-- Удаленные объекты в Axenta
SELECT * FROM axenta_object_snapshots 
WHERE axenta_deleted_at IS NOT NULL;

-- Объекты с номерами телефонов
SELECT object_name, phone_numbers 
FROM axenta_object_snapshots 
WHERE phone_numbers IS NOT NULL;
```

### Go код

```go
// Поиск по создателю
var snapshots []models.AxentaObjectSnapshot
db.Where("creator_name ILIKE ?", "%Иванов%").Find(&snapshots)

// Фильтр по дате создания
db.Where("axenta_created_at >= ?", startDate).Find(&snapshots)

// Парсинг телефонов
if snapshot.PhoneNumbers != nil {
    var phones []string
    json.Unmarshal([]byte(*snapshot.PhoneNumbers), &phones)
}
```
